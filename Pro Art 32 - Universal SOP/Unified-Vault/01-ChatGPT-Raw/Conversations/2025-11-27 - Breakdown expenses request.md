records = []
date_re = re.compile(r"Invoice date\s+(\d{4}/\d{2}/\d{2})")

def parse_invoice(pdf_path):
    try:
        doc = fitz.open(pdf_path)
    except Exception as e:
        return None
    if doc.page_count == 0:
        return None
    text = doc[0].get_text("text")
    # date
    m = date_re.search(text)
    inv_date = None
    if m:
        inv_date = m.group(1)
    # lines
    lines = [l.strip() for l in text.splitlines() if l.strip()!='']
    amt = None
    for i, line in enumerate(lines):
        if line.startswith("Total Amount"):
            # next numeric line
            for j in range(i+1, min(i+5, len(lines))):
                ln = lines[j]
                # number potentially with currency etc
                m2 = re.search(r"(-?\d+\.\d{2})", ln.replace(",",""))
                if m2:
                    amt = float(m2.group(1))
                    break
    if inv_date and amt is not None:
        return inv_date, amt
    return None

for path in pdfs:
    res = parse_invoice(path)
    if res:
        records.append({"file": os.path.basename(path), "date_str": res[0], "amount": res[1]})
